# Kiwi 三轮全向底盘 ROS 2 功能包的构建入口，安装驱动、遥操作、参数和启动文件。
cmake_minimum_required(VERSION 3.8)
project(kiwi_base)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
include_directories(include)
add_library(kiwi_base_core src/kiwi_kinematics.cpp src/robomaster_protocol.cpp)
ament_target_dependencies(kiwi_base_core geometry_msgs)
add_executable(kiwi_base_node src/kiwi_base_node.cpp)
target_link_libraries(kiwi_base_node kiwi_base_core)
ament_target_dependencies(kiwi_base_node rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs diagnostic_msgs)
add_executable(kiwi_teleop_keyboard src/kiwi_teleop_keyboard.cpp)
ament_target_dependencies(kiwi_teleop_keyboard rclcpp geometry_msgs)
install(TARGETS kiwi_base_node kiwi_teleop_keyboard kiwi_base_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
install(DIRECTORY launch/ DESTINATION share/${PROJECT_NAME}/launch)
# 安装命令发送工具和 0x3B 组合反馈接收工具，便于脱离 ROS 节点进行串口联调。
install(
  PROGRAMS tools/kiwi_serial_test.py tools/kiwi_feedback_receiver.py
  DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_kinematics test/test_kinematics.cpp)
  target_link_libraries(test_kinematics kiwi_base_core)
  ament_target_dependencies(test_kinematics geometry_msgs)
  ament_add_gtest(test_protocol test/test_protocol.cpp)
  target_link_libraries(test_protocol kiwi_base_core)
  ament_target_dependencies(test_protocol geometry_msgs)
endif()
ament_package()
