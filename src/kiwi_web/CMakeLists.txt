cmake_minimum_required(VERSION 3.8)
project(kiwi_web)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/kiwi_web_server scripts/kiwi_web_install_service DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config web systemd DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_web test TIMEOUT 120)
endif()
ament_package()
