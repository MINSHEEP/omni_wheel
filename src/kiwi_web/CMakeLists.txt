cmake_minimum_required(VERSION 3.8)
project(kiwi_web)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
# symlink-install 使用构建目录中的可执行副本，避免源脚本权限丢失后 ros2 run 找不到入口。
set(KIWI_WEB_PROGRAMS)
foreach(program kiwi_web_server kiwi_web_install_service)
  set(installed_program "${CMAKE_CURRENT_BINARY_DIR}/${program}")
  configure_file("scripts/${program}" "${installed_program}" COPYONLY)
  file(CHMOD "${installed_program}" PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  list(APPEND KIWI_WEB_PROGRAMS "${installed_program}")
endforeach()
install(PROGRAMS ${KIWI_WEB_PROGRAMS} DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config web systemd DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_web test TIMEOUT 120)
endif()
ament_package()
